package com.platform.security;

import com.platform.config.SecurityConfig;
import com.platform.service.SecureMediaService;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.security.test.context.support.WithMockUser;

import static org.junit.jupiter.api.Assertions.*;

@SpringBootTest
class RoleAccessTest {

    @Autowired
    private SecureMediaService secureMediaService;

    @Test
    @WithMockUser(roles = "USER")
    void testUserRoleAccess() {
        // USER должен иметь доступ к stream
        assertTrue(hasAccessToStream("USER"));
        
        // USER не должен иметь доступ к upload
        assertFalse(hasAccessToUpload("USER"));
        
        // USER не должен иметь доступ к download
        assertFalse(canDownloadOriginal("USER"));
    }

    @Test
    @WithMockUser(roles = "PREMIUM")
    void testPremiumRoleAccess() {
        // PREMIUM должен иметь доступ к stream
        assertTrue(hasAccessToStream("PREMIUM"));
        
        // PREMIUM не должен иметь доступ к upload
        assertFalse(hasAccessToUpload("PREMIUM"));
        
        // PREMIUM не должен иметь доступ к download
        assertFalse(canDownloadOriginal("PREMIUM"));
    }

    @Test
    @WithMockUser(roles = "ADVANCED")
    void testAdvancedRoleAccess() {
        // ADVANCED должен иметь доступ к stream
        assertTrue(hasAccessToStream("ADVANCED"));
        
        // ADVANCED не должен иметь доступ к upload
        assertFalse(hasAccessToUpload("ADVANCED"));
        
        // ADVANCED не должен иметь доступ к download
        assertFalse(canDownloadOriginal("ADVANCED"));
    }

    @Test
    @WithMockUser(roles = "CREATOR")
    void testCreatorRoleAccess() {
        // CREATOR должен иметь доступ к stream
        assertTrue(hasAccessToStream("CREATOR"));
        
        // CREATOR должен иметь доступ к upload
        assertTrue(hasAccessToUpload("CREATOR"));
        
        // CREATOR не должен иметь доступ к download
        assertFalse(canDownloadOriginal("CREATOR"));
    }

    @Test
    @WithMockUser(roles = "MODERATOR_ADMIN")
    void testModeratorAdminRoleAccess() {
        // MODERATOR_ADMIN должен иметь доступ к stream
        assertTrue(hasAccessToStream("MODERATOR_ADMIN"));
        
        // MODERATOR_ADMIN должен иметь доступ к upload
        assertTrue(hasAccessToUpload("MODERATOR_ADMIN"));
        
        // MODERATOR_ADMIN должен иметь доступ к admin
        assertTrue(hasAccessToAdmin("MODERATOR_ADMIN"));
        
        // MODERATOR_ADMIN не должен иметь доступ к download
        assertFalse(canDownloadOriginal("MODERATOR_ADMIN"));
    }

    @Test
    @WithMockUser(roles = "NUTRITION_ADMIN")
    void testNutritionAdminRoleAccess() {
        // NUTRITION_ADMIN должен иметь доступ к stream
        assertTrue(hasAccessToStream("NUTRITION_ADMIN"));
        
        // NUTRITION_ADMIN должен иметь доступ к upload
        assertTrue(hasAccessToUpload("NUTRITION_ADMIN"));
        
        // NUTRITION_ADMIN не должен иметь доступ к download
        assertFalse(canDownloadOriginal("NUTRITION_ADMIN"));
    }

    @Test
    @WithMockUser(roles = "SPORT_ADMIN")
    void testSportAdminRoleAccess() {
        // SPORT_ADMIN должен иметь доступ к stream
        assertTrue(hasAccessToStream("SPORT_ADMIN"));
        
        // SPORT_ADMIN должен иметь доступ к upload
        assertTrue(hasAccessToUpload("SPORT_ADMIN"));
        
        // SPORT_ADMIN не должен иметь доступ к download
        assertFalse(canDownloadOriginal("SPORT_ADMIN"));
    }

    @Test
    @WithMockUser(roles = "OWNER")
    void testOwnerRoleAccess() {
        // OWNER должен иметь доступ к stream
        assertTrue(hasAccessToStream("OWNER"));
        
        // OWNER должен иметь доступ к upload
        assertTrue(hasAccessToUpload("OWNER"));
        
        // OWNER должен иметь доступ к admin
        assertTrue(hasAccessToAdmin("OWNER"));
        
        // OWNER не должен иметь доступ к download (только SUPER_ADMIN)
        assertFalse(canDownloadOriginal("OWNER"));
    }

    @Test
    @WithMockUser(roles = "SUPER_ADMIN")
    void testSuperAdminRoleAccess() {
        // SUPER_ADMIN должен иметь доступ к stream
        assertTrue(hasAccessToStream("SUPER_ADMIN"));
        
        // SUPER_ADMIN должен иметь доступ к upload
        assertTrue(hasAccessToUpload("SUPER_ADMIN"));
        
        // SUPER_ADMIN должен иметь доступ к admin
        assertTrue(hasAccessToAdmin("SUPER_ADMIN"));
        
        // SUPER_ADMIN должен иметь доступ к download
        assertTrue(canDownloadOriginal("SUPER_ADMIN"));
    }

    // Вспомогательные методы
    private boolean hasAccessToStream(String role) {
        return switch (role) {
            case "USER", "PREMIUM", "ADVANCED", "CREATOR", 
                 "MODERATOR_ADMIN", "NUTRITION_ADMIN", "SPORT_ADMIN", 
                 "OWNER", "SUPER_ADMIN" -> true;
            default -> false;
        };
    }

    private boolean hasAccessToUpload(String role) {
        return switch (role) {
            case "CREATOR", "MODERATOR_ADMIN", "NUTRITION_ADMIN", 
                 "SPORT_ADMIN", "OWNER", "SUPER_ADMIN" -> true;
            default -> false;
        };
    }

    private boolean hasAccessToAdmin(String role) {
        return switch (role) {
            case "MODERATOR_ADMIN", "OWNER", "SUPER_ADMIN" -> true;
            default -> false;
        };
    }

    private boolean canDownloadOriginal(String role) {
        return "SUPER_ADMIN".equals(role);
    }
}

describe('Role Access Tests', () => {
  beforeEach(() => {
    // Очищаем localStorage перед каждым тестом
    cy.clearLocalStorage();
  });

  describe('USER role', () => {
    beforeEach(() => {
      cy.setDemoRole('USER');
      cy.visit('/');
    });

    it('should show limited navigation for USER', () => {
      // USER должен видеть только определенные вкладки
      cy.get('[data-tab="help"]').should('be.visible');
      cy.get('[data-tab="profile"]').should('be.visible');
      
      // USER не должен видеть другие вкладки или они должны быть заблокированы
      cy.get('[data-tab="main"]').click();
      cy.get('.access-denied-message').should('be.visible');
    });

    it('should not show chat switching for USER', () => {
      // USER не должен видеть переключатель чатов
      cy.get('.chat-split').should('not.exist');
    });

    it('should show access denied for restricted tabs', () => {
      cy.get('[data-tab="archive"]').click();
      cy.get('.access-denied-message').should('be.visible');
      
      cy.get('[data-tab="scheduled"]').click();
      cy.get('.access-denied-message').should('be.visible');
      
      cy.get('[data-tab="dashboard"]').click();
      cy.get('.access-denied-message').should('be.visible');
    });
  });

  describe('PREMIUM role', () => {
    beforeEach(() => {
      cy.setDemoRole('PREMIUM');
      cy.visit('/');
    });

    it('should show full navigation for PREMIUM', () => {
      // PREMIUM должен видеть все вкладки

      cy.setDemoRole('ADVANCED');
      cy.visit('/');
    });

    it('should show full navigation for ADVANCED', () => {

      cy.get('[data-tab="scheduled"]').should('be.visible');
      cy.get('[data-tab="dashboard"]').should('be.visible');
      cy.get('[data-tab="chat"]').should('be.visible');
      cy.get('[data-tab="help"]').should('be.visible');
      cy.get('[data-tab="profile"]').should('be.visible');
    });
  });

  describe('CREATOR role', () => {
    beforeEach(() => {
      cy.setDemoRole('CREATOR');
      cy.visit('/');
    });

    it('should show upload capability for CREATOR', () => {
      // CREATOR должен видеть кнопку загрузки
      cy.get('[data-testid="upload-button"]').should('be.visible');
    });
  });

  describe('MODERATOR_ADMIN role', () => {
    beforeEach(() => {
      cy.setDemoRole('MODERATOR_ADMIN');
      cy.visit('/');
    });

    it('should show admin panel for MODERATOR_ADMIN', () => {
      // MODERATOR_ADMIN должен видеть админ-панель
      cy.get('[data-testid="admin-panel"]').should('be.visible');

      cy.get('[data-testid="moderation-tools"]').should('be.visible');

## 🚀 Запуск тестов

### 1. Unit и Integration тесты (бэкенд)

```bash
cd backend

# Запуск всех тестов
mvn test

# Запуск только тестов ролей
mvn test -Dtest=RoleAccessTest
mvn test -Dtest=SecureMediaServiceTest
mvn test -Dtest=*IntegrationTest

# Запуск с отчетом
mvn test surefire-report:report
```

### 2. E2E тесты (фронтенд)

```bash
cd frontend

# Установка Cypress (если не установлен)
npm install cypress --save-dev

# Запуск Cypress в интерактивном режиме
npx cypress open

# Запуск всех тестов в headless режиме
npx cypress run

# Запуск только тестов ролей
npx cypress run --spec "cypress/e2e/role-access.cy.js"
```

### 3. Ручное тестирование

```bash
# Запуск бэкенда
cd backend
mvn spring-boot:run

# Запуск фронтенда
cd frontend
npm start
```
