private String getUserRole(Jwt jwt) {
    List<String> roles = jwt.getClaimAsStringList("realm_access.roles");
    if (roles == null) return "USER";
    if (roles.contains("MODERATOR_ADMIN")) return "MODERATOR_ADMIN";
    if (roles.contains("SUPER_ADMIN")) return "SUPER_ADMIN";
    if (roles.contains("OWNER")) return "OWNER";
    if (roles.contains("CREATOR")) return "CREATOR";
    if (roles.contains("SPORT_ADMIN")) return "SPORT_ADMIN";
    if (roles.contains("NUTRITION_ADMIN")) return "NUTRITION_ADMIN";
    if (roles.contains("ADVANCED")) return "ADVANCED";
    if (roles.contains("PREMIUM")) return "PREMIUM";
    return "USER";
}
```

public class RoleUtils {
    public static boolean isAdmin(String role) {
        return MODERATOR_ADMIN.equals(role) || 
               SUPER_ADMIN.equals(role) || 
               OWNER.equals(role);
    }
    
    public static boolean canUploadMedia(String role) {
        return CREATOR.equals(role) || 
               MODERATOR_ADMIN.equals(role) || 
               SUPER_ADMIN.equals(role) || 
               NUTRITION_ADMIN.equals(role) || 
               SPORT_ADMIN.equals(role) || 
               OWNER.equals(role);
    }
    
    public static boolean canDownloadOriginal(String role) {
        return SUPER_ADMIN.equals(role);
    }
}
```
